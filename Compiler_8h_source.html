59 Compiler(const char* text, Expression<T, N>& out, const Resolver& resolver)

61 errPos_(0), depth_(0), stack_(0), maxStack_(0) {
62 cur_.type = Tok::End;

64 cur_.number = T();

70 Error run(size_t& errorPos) {
71 out_.clear();
72 bool ok = advance();
73 if (ok && cur_.type == Tok::End) ok = fail(Error::Empty, 0);
74 if (ok) ok = parseExpression();
75 // unit conversion: "expr to unit" expresses the value in that unit
76 if (ok && cur_.type == Tok::Ident && isToKeyword()) {
77 ok = advance();
78 if (ok && cur_.type != Tok::Ident) {
79 ok = fail(cur_.type == Tok::End ? Error::UnexpectedEnd : Error::Syntax, cur_.pos);
80 }
81 if (ok) {
82 T factor = T();
83 if (!lookupConstant<T>(cur_.name, factor)) {
84 ok = fail(Error::UnknownIdentifier, cur_.pos);
85 } else {
86 ok = emitPush(factor) && emitBinary(OpCode::Div) && advance();
87 }
88 }
89 }
90 if (ok && cur_.type != Tok::End) {
91 ok = fail(cur_.type == Tok::RParen ? Error::UnbalancedParenthesis
92 : Error::UnexpectedToken,
93 cur_.pos);
94 }
95 if (!ok) {
96 errorPos = errPos_;
97 return err_;
98 }
99 out_.setMaxStack(static_cast<size_t>(maxStack_));
100 errorPos = 0;
101 return Error::None;
102 }

105 // ---- error handling -------------------------------------------------
106 bool fail(Error e, size_t pos) {
107 if (err_ == Error::None) { // keep the first error
108 err_ = e;
109 errPos_ = pos;

111 return false;

114 // ---- lexer ----------------------------------------------------------
115 bool advance() {
116 while (isSpace(src_[pos_])) ++pos_;
117 cur_.pos = pos_;
118 char c = src_[pos_];
119 if (c == '\0') {
120 cur_.type = Tok::End;
121 return true;
122 }
123 if (isDigit(c) || (c == '.' && isDigit(src_[pos_ + 1]))) return scanNumber();
124 if (isIdentStart(c)) return scanIdentifier();
125 ++pos_;
126 switch (c) {
127 case '+': cur_.type = Tok::Plus; return true;
128 case '-': cur_.type = Tok::Minus; return true;
129 case '/': cur_.type = Tok::Slash; return true;
130 case '%': cur_.type = Tok::Percent; return true;
131 case '^': cur_.type = Tok::Caret; return true;
132 case '!': cur_.type = Tok::Bang; return true;
133 case '(': cur_.type = Tok::LParen; return true;
134 case ')': cur_.type = Tok::RParen; return true;
135 case ',': cur_.type = Tok::Comma; return true;
136 case '*': // '**' is accepted as an alias for '^'
137 if (src_[pos_] == '*') {
138 ++pos_;
139 cur_.type = Tok::Caret;
140 } else {
141 cur_.type = Tok::Star;
142 }
143 return true;
144 default:
145 return fail(Error::Syntax, cur_.pos);
146 }
147 }

149 bool scanNumber() {
150 size_t start = pos_;
151 while (isDigit(src_[pos_])) ++pos_;
152 if (src_[pos_] == '.') {
153 ++pos_;
154 while (isDigit(src_[pos_])) ++pos_;
155 }
156 if (src_[pos_] == 'e' || src_[pos_] == 'E') { // exponent only if digits follow
157 size_t p = pos_ + 1;
158 if (src_[p] == '+' || src_[p] == '-') ++p;
159 if (isDigit(src_[p])) {
160 pos_ = p;
161 while (isDigit(src_[pos_])) ++pos_;
162 }
163 }
164 size_t len = pos_ - start;
165 char buf[32];
166 if (len >= sizeof(buf)) return fail(Error::Syntax, start);
167 memcpy(buf, src_ + start, len);
168 buf[len] = '\0';
169 cur_.type = Tok::Number;
170 cur_.number = static_cast<T>(strtod(buf, nullptr));
171 return true;
172 }
173
174 bool scanIdentifier() {
175 size_t start = pos_;
176 while (isIdentPart(src_[pos_])) ++pos_;
177 size_t len = pos_ - start;
178 if (len > TINYCALC_MAX_NAME) return fail(Error::UnknownIdentifier, start);
179 memcpy(cur_.name, src_ + start, len);
180 cur_.name[len] = '\0';
181
182 size_t p = pos_;
183 while (isSpace(src_[p])) ++p;
184 if (src_[p] == '(') { // identifier followed by '(' must be a function
185 cur_.function = findFunction(cur_.name); // nullptr: a user function
186 cur_.builtin = cur_.function != nullptr;
187 cur_.type = Tok::Function;
188 } else {
189 cur_.type = Tok::Ident;
190 }
191 return true;
192 }

194 // ---- parser ---------------------------------------------------------
195 bool enter() {
196 if (++depth_ > TINYCALC_MAX_NESTING) return fail(Error::NestingTooDeep, cur_.pos);
197 return true;
198 }
199 void leave() { --depth_; }

201 // the keyword of a unit conversion ("5 km to m")
202 bool isToKeyword() const { return cur_.type == Tok::Ident && strcmp(cur_.name, "to") == 0; }
203
204 static bool startsOperand(Tok t) {
205 return t == Tok::Number || t == Tok::Ident || t == Tok::Function || t == Tok::LParen;
206 }

208 bool parseExpression() {
209 if (!enter()) return false;
210 if (!parseTerm()) return false;
211 while (cur_.type == Tok::Plus || cur_.type == Tok::Minus) {
212 OpCode op = cur_.type == Tok::Plus ? OpCode::Add : OpCode::Sub;
213 if (!advance() || !parseTerm() || !emitBinary(op)) return false;
214 }
215 leave();
216 return true;
217 }
218
219 bool parseTerm() {
220 if (!parseUnary()) return false;
221 for (;;) {
222 OpCode op = OpCode::Mul;
223 if (cur_.type == Tok::Star) {
224 op = OpCode::Mul;
225 } else if (cur_.type == Tok::Slash) {
226 op = OpCode::Div;
227 } else if (cur_.type == Tok::Percent) {
228 op = OpCode::Mod;
229 } else if (startsOperand(cur_.type) && !isToKeyword()) { // implicit multiplication: 2pi, 3(x+1)
230 if (!parseUnary() || !emitBinary(OpCode::Mul)) return false;
231 continue;
232 } else {
233 return true;
234 }
235 if (!advance() || !parseUnary() || !emitBinary(op)) return false;
236 }
237 }
238
239 bool parseUnary() {
240 if (cur_.type == Tok::Minus || cur_.type == Tok::Plus) {
241 bool negate = cur_.type == Tok::Minus;
242 if (!enter() || !advance() || !parseUnary()) return false;
243 leave();
244 return !negate || emitUnary(OpCode::Neg);
245 }
246 return parsePower();
247 }
248
249 bool parsePower() {
250 if (!parsePostfix()) return false;
251 if (cur_.type == Tok::Caret) {
252 if (!advance() || !parseUnary() || !emitBinary(OpCode::Pow)) return false;
253 }
254 return true;
255 }
256
257 bool parsePostfix() {
258 if (!parsePrimary()) return false;
259 while (cur_.type == Tok::Bang) {
260 if (!advance() || !emitCall(Fn::Fact, 1)) return false;
261 }
262 return true;
263 }

265 bool parsePrimary() {
266 switch (cur_.type) {
267 case Tok::Number:
268 return emitPush(cur_.number) && advance();
269 case Tok::Ident:
270 return parseIdentifier();
271 case Tok::Function:
272 return parseCall();
273 case Tok::LParen: {
274 size_t open = cur_.pos;
275 if (!advance() || !parseExpression()) return false;
276 if (cur_.type != Tok::RParen) {
277 if (cur_.type == Tok::End) return fail(Error::UnbalancedParenthesis, open);
278 return fail(Error::UnexpectedToken, cur_.pos);
279 }
280 return advance();
281 }
282 case Tok::End:
283 return fail(Error::UnexpectedEnd, cur_.pos);
284 default:
285 return fail(Error::UnexpectedToken, cur_.pos);
286 }
287 }

289 bool parseIdentifier() {
290 T constant = T();
291 if (lookupConstant<T>(cur_.name, constant)) return emitPush(constant) && advance();
292 int slot = res_.find(cur_.name);
293 if (slot < 0) return fail(Error::UnknownIdentifier, cur_.pos);
294 return emit(OpCode::Load, static_cast<uint8_t>(slot), 0, T(), 1) && advance();
295 }
296
297 bool parseCall() {
298 const FunctionInfo* info = cur_.function;
299 size_t start = cur_.pos;
300 char name[TINYCALC_MAX_NAME + 1];
301 strcpy(name, cur_.name);
302 if (!advance() || !advance()) return false; // skip name and '('
303 int argc = 0;
304 if (cur_.type != Tok::RParen) {
305 for (;;) {
306 if (!parseExpression()) return false;
307 ++argc;
308 if (cur_.type != Tok::Comma) break;
309 if (!advance()) return false;
310 }
311 }
312 if (cur_.type != Tok::RParen) {
313 if (cur_.type == Tok::End) return fail(Error::UnbalancedParenthesis, start);
314 return fail(Error::UnexpectedToken, cur_.pos);
315 }
316 if (!advance()) return false;
317 if (info != nullptr) { // built-in function
318 if (argc < info->minArgs || argc > info->maxArgs) {
319 return fail(Error::WrongArgumentCount, start);
320 }
321 return emitCall(info->id, argc);
322 }
323 // user function: looked up by name, the number of parameters must match
324 int slot = -1, params = 0;
325 if (!res_.findUserFunction(name, slot, params)) return fail(Error::UnknownFunction, start);
326 if (argc != params) return fail(Error::WrongArgumentCount, start);
327 return emit(OpCode::UserCall, static_cast<uint8_t>(slot), static_cast<uint8_t>(argc),
328 T(), 1 - argc);
329 }

331 // ---- code emission --------------------------------------------------
332 bool emit(OpCode code, uint8_t index, uint8_t argc, T value, int stackDelta) {
333 Op<T> op;
334 op.code = code;
335 op.index = index;
336 op.argc = argc;
337 op.value = value;
338 if (!out_.append(op)) return fail(Error::TooComplex, cur_.pos);
339 stack_ += stackDelta;
340 if (stack_ > maxStack_) maxStack_ = stack_;
341 return true;
342 }
343 bool emitPush(T v) { return emit(OpCode::Push, 0, 0, v, 1); }
344 bool emitBinary(OpCode code) { return emit(code, 0, 0, T(), -1); }
345 bool emitUnary(OpCode code) { return emit(code, 0, 0, T(), 0); }
346 bool emitCall(Fn fn, int argc) {
347 return emit(OpCode::Call, static_cast<uint8_t>(fn), static_cast<uint8_t>(argc),
348 T(), 1 - argc);
349 }
