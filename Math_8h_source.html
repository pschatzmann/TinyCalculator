134 // Gamma function. Caller must exclude non-positive integers.
135 // Reflection for x < 0.5; otherwise the argument is shifted up to >= 15
136 // using gamma(x + 1) = x * gamma(x), and the Stirling series for ln(gamma)
137 // is evaluated (absolute error ~1e-14 in double precision).
138 static T gamma(T x) {
139 if (x < c(0.5)) return pi() / (P::sin(pi() * x) * gamma(c(1) - x));
140 T shift = c(1);
141 while (x < c(15)) {
142 shift /= x;
143 x += c(1);
144 }
145 T z = x;
146 T z2 = z * z;
147 T lnG = (z - c(0.5)) * P::log(z) - z + c(0.918938533204672741780329736406); // ln(sqrt(2pi))
148 lnG += (c(1) / c(12)) / z
149 - (c(1) / c(360)) / (z * z2)
150 + (c(1) / c(1260)) / (z * z2 * z2)
151 - (c(1) / c(1680)) / (z * z2 * z2 * z2);
152 return P::exp(lnG) * shift;
153 }
154
155 // Error function. Uses a series with only positive terms, so there is no
156 // cancellation; for |x| > 6 the result is +-1 to working precision.
157 static T erf(T x) {
158 if (x == c(0)) return c(0);
159 T ax = absolute(x);
160 if (ax > c(6)) return x < c(0) ? c(-1) : c(1);
161 const T x2 = c(2) * ax * ax;
162 T term = ax, sum = ax;
163 for (int n = 1; n < 400; ++n) {
164 term *= x2 / c(2 * n + 1);
165 sum += term;
166 if (term <= sum * c(1e-9)) break;
167 }
168 T r = c(1.1283791670955126) * P::exp(-ax * ax) * sum; // 2/sqrt(pi)
169 return x < c(0) ? -r : r;
170 }
171 static T erfc(T x) { return c(1) - erf(x); }
