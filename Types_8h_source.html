59// The text of an error code. On AVR the pointer refers to flash: print it with
60// errorTextFlash(), or copy it with strcpy_P().
61inline const char* errorText(Error e) {
62 const size_t i = static_cast<size_t>(e);
63 if (i >= sizeof(kErrorTexts) / sizeof(kErrorTexts[0])) {
64 return TINYCALC_READ_PTR(kUnknownError);
65 }
66 return TINYCALC_READ_PTR(kErrorTexts[i]);
67}

86// true if s is a valid identifier that fits into TINYCALC_MAX_NAME characters
87inline bool isIdentifier(const char* s) {
88 if (!isIdentStart(s[0])) return false;
89 size_t len = 0;
90 while (isIdentPart(s[len])) ++len;
