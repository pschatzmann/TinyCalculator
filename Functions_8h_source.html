73// Copies entry i of the function table into out. False if i is out of range.
74inline bool functionAt(size_t i, FunctionInfo& out) {
75 size_t count = 0;
76 const FunctionInfo* table = functionTable(count);
77 if (i >= count) return false;
78 TINYCALC_MEMCPY_P(&out, &table[i], sizeof(FunctionInfo));
79 return true;
80}

82// Looks up a built-in function by name; copies its entry into out.
83inline bool findFunction(const char* name, FunctionInfo& out) {
84 size_t count = 0;
85 functionTable(count);
86 for (size_t i = 0; i < count; ++i) {
87 functionAt(i, out);
88 if (strcmp(out.name, name) == 0) return true;
89 }
90 return false;
91}

118// Looks up a unit by name; returns its factor, or false.
119inline bool findUnitFactor(const char* name, double& factor) {
120 size_t count = 0;
121 const UnitInfo* table = unitTable(count);
122 for (size_t i = 0; i < count; ++i) {
123 UnitInfo unit;
124 TINYCALC_MEMCPY_P(&unit, &table[i], sizeof(UnitInfo));
125 if (strcmp(unit.name, name) == 0) {
126 factor = unit.factor;
127 return true;
128 }
129 }
130 return false;

133// Temperature units: they are not linear (0 degC is not 0 K), so they are
134// converted with "value unit to unit" rather than with a factor.
135// Index: 0 degC, 1 degF, 2 K; -1 if the name is not a temperature unit.

141inline int temperatureIndex(const char* name) {

156// Looks up a built-in constant (pi, e, tau, phi) or unit (km, s, ...).
157template <typename T>
158inline bool lookupConstant(const char* name, T& value) {
159 typedef Math<T> M;
160 struct Constant {
161 char name[4];
162 uint8_t index;
163 };
164 static const Constant constants[] TINYCALC_PROGMEM = {
165 {"pi", 0}, {"e", 1}, {"tau", 2}, {"phi", 3}};
166 for (size_t i = 0; i < sizeof(constants) / sizeof(constants[0]); ++i) {
167 Constant c;
168 TINYCALC_MEMCPY_P(&c, &constants[i], sizeof(Constant));
169 if (strcmp(c.name, name) == 0) {
170 switch (c.index) {
171 case 0: value = M::pi(); break;
172 case 1: value = M::e(); break;
173 case 2: value = M::tau(); break;
174 default: value = M::phi(); break;
175 }
176 return true;
177 }
178 }
179 double factor = 0.0;
180 if (findUnitFactor(name, factor)) {
181 value = static_cast<T>(factor);
182 return true;
183 }
184 return false;
185}
