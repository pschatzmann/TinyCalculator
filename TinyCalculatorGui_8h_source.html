192 void press(const char* action) {
193 if (action == nullptr || *action == '\0') return;
194 std::string a = action;
195 error_.clear();
196
197 if (plotting_) {
198 handlePlotKey(a);
199 refresh();
200 return;
201 }
202
203 if (a == "page") {
204 changePage(1);
205 } else if (a == "plot") {
206 plotExpression();
207 } else if (a == "angle") {
208 toggleAngle();
209 } else if (a == "clear") {
210 expr_.clear();
211 showResult_ = false;
212 } else if (a == "del") {
213 if (showResult_) {
214 expr_.clear();
215 showResult_ = false;
216 } else if (!expr_.empty()) {
217 expr_.erase(expr_.size() - 1);
218 }
219 } else if (a == "=") {
220 evaluate();
221 } else {
222 // digits, operators, functions and constants
223 // after "=", an operator continues with the result, anything else starts over
224 std::string prefix = expr_;
225 if (showResult_) {
226 bool continues = a == "+" || a == "-" || a == "*" || a == "/" || a == "^" ||
227 a == "%" || a == "!" || a == ",";
228 prefix = continues ? "ans" : "";
229 }
230 if (!acceptable(a, prefix)) { // invalid combination: the key is ignored
231 refresh();
232 return;
233 }
234 if (showResult_) {
235 expr_ = prefix;
236 showResult_ = false;
237 }
238 if (needsProduct(expr_, a)) expr_ += "*";
239 expr_ += a;
240 }
241 refresh();
242 }

244 // ---- key entry rules --------------------------------------------------
245 // The keypad only accepts keys that continue the expression validly, e.g.
246 // no operator after an operator, no ")" without "(", no second "." in a
247 // number. Rejected keys are ignored.
248
249 static char lastChar(const std::string& t) { return t.empty() ? '\0' : t[t.size() - 1]; }
250 static bool isDigitChar(char c) { return c >= '0' && c <= '9'; }
251 static bool isLetter(char c) { return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z'); }
252 static bool isOperatorChar(char c) {
253 return c == '+' || c == '-' || c == '*' || c == '/' || c == '^' || c == '%' || c == ',';
254 }
255 static bool isBinaryOp(const std::string& a) {
256 return a == "+" || a == "*" || a == "/" || a == "^" || a == "%";
257 }
258 static bool isNameKey(const std::string& a) {
259 return a == "pi" || a == "e" || a == "tau" || a == "phi" || a == "ans" || a == "x";
260 }
261 static bool isFunctionKey(const std::string& a) { return !a.empty() && a[a.size() - 1] == '('; }
262
263 // number of '(' not yet closed
264 static int openDepth(const std::string& t) {
265 int depth = 0;
266 for (char c : t) {
267 if (c == '(') ++depth;
268 if (c == ')' && depth > 0) --depth;
269 }
270 return depth;
271 }
272
273 // true if the number being typed already contains a '.'
274 static bool numberHasDot(const std::string& t) {
275 for (size_t i = t.size(); i-- > 0;) {
276 if (t[i] == '.') return true;
277 if (!isDigitChar(t[i])) return false;
278 }
279 return false;
280 }
281
282 // true if key `a` may follow the expression `t`
283 static bool acceptable(const std::string& a, const std::string& t) {
284 const char c = lastChar(t);
285 const bool empty = t.empty();
286 const bool open = c == '(';
287 const bool op = isOperatorChar(c);
288 const bool operand = !empty && !open && !op; // a number, name, ")" or "!"
289
290 if (a == "(" || isFunctionKey(a)) return true; // a function is also a call start
291 if (a == ")") return openDepth(t) > 0 && operand;
292 if (a == "!") return operand && c != '.';
293 if (a == ",") return openDepth(t) > 0 && operand;
294 if (a == "-") return c != '-'; // subtraction or a sign, but never "--"
295 if (a == ".") return !numberHasDot(t) && !isLetter(c) && c != ')' && c != '!';
296 if (isBinaryOp(a)) return operand;
297 if (isNameKey(a)) return c != '.';
298 return true; // digits
299 }
300
301 // an explicit "*" is needed when a name, ")" or "!" is followed by an operand
302 // (so that "pi" then "2" is "pi*2", not the name "pi2")
303 static bool needsProduct(const std::string& t, const std::string& a) {
304 const char c = lastChar(t);
305 const bool endsOperand = isLetter(c) || c == ')' || c == '!';
306 const bool startsOperand = isDigitChar(a[0]) || isLetter(a[0]) || a[0] == '(';
307 return endsOperand && startsOperand;
308 }

458 // Positions all widgets for the current size (bounds_) and mode.
459 //
460 // Wide areas (landscape) put the display on the left and the keys on the
461 // right; tall areas (portrait) stack the display above the keys. The keys
462 // use the space the display does not need, so the keypad fills the area.
463 void relayout() {
464 panel_.bounds = bounds_;
465 const int32_t x = bounds_.x, y = bounds_.y, w = bounds_.w, h = bounds_.h;
466 const bool landscape = w > h;
467 tinymd::Bounds display, pad;
468 if (landscape) {
469 const int32_t dw = (w - 2 * kMargin) / 2;
470 display = tinymd::Bounds(x + kMargin, y + kMargin, dw, h - 2 * kMargin);
471 pad = tinymd::Bounds(x + kMargin + dw + kGap, y + kMargin,
472 w - 2 * kMargin - dw - kGap, h - 2 * kMargin);
473 } else {
474 // the plot needs more height than the expression and result lines
475 const int32_t dh = plotting_ ? (h * 11) / 20 : (h * 3) / 10;
476 display = tinymd::Bounds(x + kMargin, y + kMargin, w - 2 * kMargin, dh);
477 pad = tinymd::Bounds(x + kMargin, y + kMargin + dh + kGap,
478 w - 2 * kMargin, h - 2 * kMargin - dh - kGap);
479 }
480
481 if (plotting_) {
482 const int32_t fh = min32(kFieldH, display.h / 4);
483 // room for the field labels ("From", "To"), which are drawn above the fields
484 const int32_t labelH = 18;
485 plot_.bounds = tinymd::Bounds(display.x, display.y, display.w,
486 display.h - fh - labelH - kGap);
487 const int32_t fw = (display.w - kGap) / 2;
488 const int32_t fy = display.y + display.h - fh;
489 fromField_.bounds = tinymd::Bounds(display.x, fy, fw, fh);
490 toField_.bounds = tinymd::Bounds(display.x + fw + kGap, fy, display.w - fw - kGap, fh);
491 placeKeys(pad, 4); // numeric pad
492 } else {
493 const int32_t eh = min32(kExprH, display.h / 3);
494 exprLabel_.bounds = tinymd::Bounds(display.x, display.y, display.w, eh);
495 resultLabel_.bounds = tinymd::Bounds(display.x, display.y + eh, display.w,
496 display.h - eh);
497 placeKeys(pad, 7); // keypad
498 }
499 }

501 static int32_t min32(int32_t a, int32_t b) { return a < b ? a : b; }

503 // Places the key buttons in a grid of `rows` rows (4 columns) inside `area`.
504 void placeKeys(tinymd::Bounds area, int32_t rows) {
505 const int32_t cols = 4;
506 int32_t keyW = (area.w - (cols - 1) * kGap) / cols;
507 int32_t keyH = (area.h - (rows - 1) * kGap) / rows;
508 if (keyW < 1) keyW = 1;
509 if (keyH < 1) keyH = 1;
510 for (size_t i = 0; i < kKeyCount; ++i) {
511 keys_[i].onClick = [this, i]() { press(activeKeys()[i].action); };
512 int32_t row = static_cast<int32_t>(i) / cols;
513 int32_t col = static_cast<int32_t>(i) % cols;
514 if (row >= rows) {
515 // not on the grid: zero size, so hidden keys do not extend the scroll area
516 keys_[i].bounds = tinymd::Bounds(bounds_.x, bounds_.y, 0, 0);
517 continue;
518 }
519 keys_[i].bounds = tinymd::Bounds(area.x + col * (keyW + kGap),
520 area.y + row * (keyH + kGap), keyW, keyH);
521 }
522 }

602 // Parses a whole number text; false if empty or not a number.
603 static bool parseNumber(const std::string& text, float& out) {
604 if (text.empty() || text == "-" || text == ".") return false;
605 char* end = nullptr;
606 out = strtof(text.c_str(), &end);
607 return end != nullptr && *end == '\0';

610 // Keys of the plot pad. The From/To fields report focus through onFocus.
611 void handlePlotKey(const std::string& a) {
612 if (a == "back") {
613 closePlot();
614 return;
615 }
616 if (a == "from") {
617 fromField_.focus();
618 return;
619 }
620 if (a == "to") {
621 toField_.focus();
622 return;
623 }
624 std::string& text = activeField_ == 0 ? fromText_ : toText_;
625 if (a == "del") {
626 if (!text.empty()) text.erase(text.size() - 1);
627 } else if (a == "-") {
628 if (text.empty()) text = "-"; // a minus sign only at the start
629 } else if (a == ".") {
630 if (text.find('.') == std::string::npos) text += ".";
631 } else if (a.size() == 1 && a[0] >= '0' && a[0] <= '9') {
632 if (text.size() < 9) text += a;
633 } else {
634 return;
635 }
636 fromField_.setText(fromText_.c_str());
637 toField_.setText(toText_.c_str());
638 applyRangeText();
639 }

641 // Redraws the plot if both fields hold a valid range (To above From).
642 void applyRangeText() {
643 float lo = 0.0f, hi = 0.0f;
644 if (!parseNumber(fromText_, lo) || !parseNumber(toText_, hi) || hi <= lo) return;
645 plotMin_ = lo;
646 plotMax_ = hi;
647 samplePlot(); // the expression compiled when the plot opened
648 }

650 // Samples the expression as a function of x over [plotMin_, plotMax_] and
651 // shows it in the chart. Points where the expression is undefined become
652 // NaN, which the chart draws as a gap.
653 Error samplePlot() {
654 typename Calc::Program prog; // compiled once, run for every sample
655 Error e = calc_.compile(expr_.c_str(), prog);
656 if (e != Error::None) return e;
657 samples_.assign(kPlotSamples, nanValue());
658 for (size_t i = 0; i < kPlotSamples; ++i) {
659 double x = plotMin_ + (plotMax_ - plotMin_) * static_cast<double>(i) /
660 static_cast<double>(kPlotSamples - 1);
661 calc_.setVariable("x", static_cast<Number>(x));
662 typename Calc::Result r = calc_.run(prog);
663 if (r.ok()) samples_[i] = static_cast<float>(r.value);
664 }
665 calc_.setVariable("x", Number(0)); // leave x at 0 for the keypad
666
667 char footer[48];
668 snprintf(footer, sizeof(footer), "x from %g to %g", static_cast<double>(plotMin_),
669 static_cast<double>(plotMax_));
670 std::string title = "f(x) = " + expr_;
671 plot_.setTitle(title.c_str());
672 plot_.setFooter(footer);
673 plot_.setXAxis(plotMin_, plotMax_, 5);
674 plot_.setValues(0, samples_);
675 return Error::None;
676 }

706 void swipeLeft() {
