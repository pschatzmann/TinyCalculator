111 void press(const char* action) {
112 if (action == nullptr || *action == '\0') return;
113 std::string a = action;
114 error_.clear();
115
116 if (plotting_) {
117 handlePlotKey(a);
118 refresh();
119 return;
120 }
121
122 if (a == "page") {
123 page_ = (page_ + 1) % kPageCount;
124 refreshKeys();
125 } else if (a == "plot") {
126 plotExpression();
127 } else if (a == "angle") {
128 toggleAngle();
129 } else if (a == "clear") {
130 expr_.clear();
131 showResult_ = false;
132 } else if (a == "del") {
133 if (showResult_) {
134 expr_.clear();
135 showResult_ = false;
136 } else if (!expr_.empty()) {
137 expr_.erase(expr_.size() - 1);
138 }
139 } else if (a == "=") {
140 evaluate();
141 } else {
142 // digits, operators, functions and constants
143 if (showResult_) {
144 // an operator after "=" continues with the result
145 bool continues = a == "+" || a == "-" || a == "*" || a == "/" || a == "^" ||
146 a == "%" || a == "!" || a == ",";
147 expr_ = continues ? "ans" : "";
148 showResult_ = false;
149 }
150 expr_ += a;
151 }
152 refresh();
153 }

264 // Position the widgets inside `b`.
265 void layout(tinymd::Bounds b) {
266 bounds_ = b;
267 const int32_t w = b.w - 2 * kMargin;
268 exprLabel_.bounds = tinymd::Bounds(b.x + kMargin, b.y + kMargin, w, kExprH);
269 resultLabel_.bounds = tinymd::Bounds(b.x + kMargin, b.y + kMargin + kExprH, w, kResultH);
270 plot_.bounds = tinymd::Bounds(b.x + kMargin, b.y + kMargin, w, kChartH);
271 const int32_t fieldW = (w - kGap) / 2;
272 fromField_.bounds = tinymd::Bounds(b.x + kMargin, fieldsY(), fieldW, kFieldH);
273 toField_.bounds = tinymd::Bounds(b.x + kMargin + fieldW + kGap, fieldsY(), fieldW, kFieldH);
274 placeKeys(b.y + kMargin + kExprH + kResultH + kGap, 7); // keypad top

279 // Places the key buttons in a grid of `rows` rows (4 columns) from `top`.
280 void placeKeys(int32_t top, int32_t rows) {
281 const int32_t cols = 4;
282 const int32_t keyW = (bounds_.w - 2 * kMargin - (cols - 1) * kGap) / cols;
283 const int32_t keyH = (bounds_.y + bounds_.h - kMargin - top - (rows - 1) * kGap) / rows;
284 for (size_t i = 0; i < kKeyCount; ++i) {
285 keys_[i].onClick = [this, i]() { press(activeKeys()[i].action); };
286 int32_t row = static_cast<int32_t>(i) / cols;
287 int32_t col = static_cast<int32_t>(i) % cols;
288 if (row >= rows) {
289 // not on the grid: zero size, so hidden keys do not extend the scroll area
290 keys_[i].bounds = tinymd::Bounds(bounds_.x, bounds_.y, 0, 0);
291 continue;
292 }
293 keys_[i].bounds = tinymd::Bounds(bounds_.x + kMargin + col * (keyW + kGap),
294 top + row * (keyH + kGap), keyW, keyH);
295 }
296 }

322 static tinymd::ButtonVariant variantFor(const char* action) {
323 std::string a = action;
324 if (a == "=") return tinymd::ButtonVariant::kFilled;
325 if (a == "angle" || a == "page" || a == "clear" || a == "del" || a == "back") {
326 return tinymd::ButtonVariant::kText;
327 }
328 if (a.size() == 1 && (a[0] >= '0' && a[0] <= '9')) return tinymd::ButtonVariant::kOutlined;
329 if (a == "." || a == "-") return tinymd::ButtonVariant::kOutlined;
330 return tinymd::ButtonVariant::kTonal;
331 }

339 void evaluate() {
340 if (expr_.empty()) return;
341 typename Calc::Result r = calc_.evaluate(expr_.c_str()); // also updates "ans"
342 if (r.ok()) {
343 resultText_ = format(r.value);
344 showResult_ = true;
345 } else {
346 showResult_ = false;
347 error_ = errorText(r.error);
348 }

351 static std::string format(Number v) {
352 // 7 significant digits for float, 12 for double
353 char buf[32];
354 if (sizeof(Number) <= 4) {
355 snprintf(buf, sizeof(buf), "%.7g", static_cast<double>(v));
356 } else {
357 snprintf(buf, sizeof(buf), "%.12g", static_cast<double>(v));
358 }
359 return buf;
360 }

363 char buf[24];
364 snprintf(buf, sizeof(buf), "%g", static_cast<double>(v));
365 return buf;

376 // Parses a whole number text; false if empty or not a number.
377 static bool parseNumber(const std::string& text, float& out) {
378 if (text.empty() || text == "-" || text == ".") return false;
379 char* end = nullptr;
380 out = strtof(text.c_str(), &end);
381 return end != nullptr && *end == '\0';
382 }

384 // Keys of the plot pad. The From/To fields report focus through onFocus.
385 void handlePlotKey(const std::string& a) {
386 if (a == "back") {
387 closePlot();
388 return;
389 }
390 if (a == "from") {
391 fromField_.focus();
392 return;
393 }
394 if (a == "to") {
395 toField_.focus();
396 return;
397 }
398 std::string& text = activeField_ == 0 ? fromText_ : toText_;
399 if (a == "del") {
400 if (!text.empty()) text.erase(text.size() - 1);
401 } else if (a == "-") {
402 if (text.empty()) text = "-"; // a minus sign only at the start
403 } else if (a == ".") {
404 if (text.find('.') == std::string::npos) text += ".";
405 } else if (a.size() == 1 && a[0] >= '0' && a[0] <= '9') {
406 if (text.size() < 9) text += a;
407 } else {
408 return;
409 }
410 fromField_.setText(fromText_.c_str());
411 toField_.setText(toText_.c_str());
412 applyRangeText();
413 }
414
415 // Redraws the plot if both fields hold a valid range (To above From).
416 void applyRangeText() {
417 float lo = 0.0f, hi = 0.0f;
418 if (!parseNumber(fromText_, lo) || !parseNumber(toText_, hi) || hi <= lo) return;
419 plotMin_ = lo;
420 plotMax_ = hi;
421 samplePlot(); // the expression compiled when the plot opened
422 }
423
424 // Samples the expression as a function of x over [plotMin_, plotMax_] and
425 // shows it in the chart. Points where the expression is undefined become
426 // NaN, which the chart draws as a gap.
427 Error samplePlot() {
428 typename Calc::Program prog; // compiled once, run for every sample
429 Error e = calc_.compile(expr_.c_str(), prog);
430 if (e != Error::None) return e;
431 samples_.assign(kPlotSamples, nanValue());
432 for (size_t i = 0; i < kPlotSamples; ++i) {
433 double x = plotMin_ + (plotMax_ - plotMin_) * static_cast<double>(i) /
434 static_cast<double>(kPlotSamples - 1);
435 calc_.setVariable("x", static_cast<Number>(x));
436 typename Calc::Result r = calc_.run(prog);
437 if (r.ok()) samples_[i] = static_cast<float>(r.value);
438 }
439 calc_.setVariable("x", Number(0)); // leave x at 0 for the keypad
440
441 char footer[48];
442 snprintf(footer, sizeof(footer), "x from %g to %g", static_cast<double>(plotMin_),
443 static_cast<double>(plotMax_));
444 std::string title = "f(x) = " + expr_;
445 plot_.setTitle(title.c_str());
446 plot_.setFooter(footer);
447 plot_.setXAxis(plotMin_, plotMax_, 5);
448 plot_.setValues(0, samples_);
449 return Error::None;
450 }

452 // Opens the plot screen for the current expression.
453 void plotExpression() {
454 if (expr_.empty()) {
455 error_ = "nothing to plot";
456 return;
457 }
458 Error e = samplePlot();
459 if (e != Error::None) {
460 error_ = errorText(e);
461 return;
462 }

464 syncRangeFields();
465 fromField_.focus(); // selects the From field (activeField_ = 0)
466 error_.clear();
467 applyMode();
468 refresh();
469 }
