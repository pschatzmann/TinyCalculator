180 void press(const char* action) {
181 if (action == nullptr || *action == '\0') return;
182 std::string a = action;
183 error_.clear();
184
185 if (plotting_) {
186 handlePlotKey(a);
187 refresh();
188 return;
189 }
190
191 if (a == "page") {
192 changePage(1);
193 } else if (a == "plot") {
194 plotExpression();
195 } else if (a == "angle") {
196 toggleAngle();
197 } else if (a == "clear") {
198 expr_.clear();
199 showResult_ = false;
200 } else if (a == "del") {
201 if (showResult_) {
202 expr_.clear();
203 showResult_ = false;
204 } else if (!expr_.empty()) {
205 expr_.erase(expr_.size() - 1);
206 }
207 } else if (a == "=") {
208 evaluate();
209 } else {
210 // digits, operators, functions and constants
211 if (showResult_) {
212 // an operator after "=" continues with the result
213 bool continues = a == "+" || a == "-" || a == "*" || a == "/" || a == "^" ||
214 a == "%" || a == "!" || a == ",";
215 expr_ = continues ? "ans" : "";
216 showResult_ = false;
217 }
218 expr_ += a;
219 }
220 refresh();
221 }

342 // Positions all widgets for the current size (bounds_) and mode.
343 //
344 // Wide areas (landscape) put the display on the left and the keys on the
345 // right; tall areas (portrait) stack the display above the keys. The keys
346 // use the space the display does not need, so the keypad fills the area.
347 void relayout() {
348 panel_.bounds = bounds_;
349 const int32_t x = bounds_.x, y = bounds_.y, w = bounds_.w, h = bounds_.h;
350 const bool landscape = w > h;
351 tinymd::Bounds display, pad;
352 if (landscape) {
353 const int32_t dw = (w - 2 * kMargin) / 2;
354 display = tinymd::Bounds(x + kMargin, y + kMargin, dw, h - 2 * kMargin);
355 pad = tinymd::Bounds(x + kMargin + dw + kGap, y + kMargin,
356 w - 2 * kMargin - dw - kGap, h - 2 * kMargin);
357 } else {
358 // the plot needs more height than the expression and result lines
359 const int32_t dh = plotting_ ? h / 2 : (h * 3) / 10;
360 display = tinymd::Bounds(x + kMargin, y + kMargin, w - 2 * kMargin, dh);
361 pad = tinymd::Bounds(x + kMargin, y + kMargin + dh + kGap,
362 w - 2 * kMargin, h - 2 * kMargin - dh - kGap);
363 }
364
365 if (plotting_) {
366 const int32_t fh = min32(kFieldH, display.h / 4);
367 plot_.bounds = tinymd::Bounds(display.x, display.y, display.w,
368 display.h - fh - kGap);
369 const int32_t fw = (display.w - kGap) / 2;
370 const int32_t fy = display.y + display.h - fh;
371 fromField_.bounds = tinymd::Bounds(display.x, fy, fw, fh);
372 toField_.bounds = tinymd::Bounds(display.x + fw + kGap, fy, display.w - fw - kGap, fh);
373 placeKeys(pad, 4); // numeric pad
374 } else {
375 const int32_t eh = min32(kExprH, display.h / 3);
376 exprLabel_.bounds = tinymd::Bounds(display.x, display.y, display.w, eh);
377 resultLabel_.bounds = tinymd::Bounds(display.x, display.y + eh, display.w,
378 display.h - eh);
379 placeKeys(pad, 7); // keypad
380 }
381 }

385 // Places the key buttons in a grid of `rows` rows (4 columns) inside `area`.
386 void placeKeys(tinymd::Bounds area, int32_t rows) {
387 const int32_t cols = 4;
388 int32_t keyW = (area.w - (cols - 1) * kGap) / cols;
389 int32_t keyH = (area.h - (rows - 1) * kGap) / rows;
390 if (keyW < 1) keyW = 1;
391 if (keyH < 1) keyH = 1;
392 for (size_t i = 0; i < kKeyCount; ++i) {
393 keys_[i].onClick = [this, i]() { press(activeKeys()[i].action); };
394 int32_t row = static_cast<int32_t>(i) / cols;
395 int32_t col = static_cast<int32_t>(i) % cols;
396 if (row >= rows) {
397 // not on the grid: zero size, so hidden keys do not extend the scroll area
398 keys_[i].bounds = tinymd::Bounds(bounds_.x, bounds_.y, 0, 0);
399 continue;
400 }
401 keys_[i].bounds = tinymd::Bounds(area.x + col * (keyW + kGap),
402 area.y + row * (keyH + kGap), keyW, keyH);
403 }
404 }

430 static tinymd::ButtonVariant variantFor(const char* action) {
431 std::string a = action;
432 if (a == "=") return tinymd::ButtonVariant::kFilled;
433 if (a == "angle" || a == "page" || a == "clear" || a == "del" || a == "back") {
434 return tinymd::ButtonVariant::kText;

436 if (a.size() == 1 && (a[0] >= '0' && a[0] <= '9')) return tinymd::ButtonVariant::kOutlined;
437 if (a == "." || a == "-") return tinymd::ButtonVariant::kOutlined;
438 return tinymd::ButtonVariant::kTonal;

459 static std::string format(Number v) {
460 // 7 significant digits for float, 12 for double
461 char buf[32];
462 if (sizeof(Number) <= 4) {
463 snprintf(buf, sizeof(buf), "%.7g", static_cast<double>(v));
464 } else {
465 snprintf(buf, sizeof(buf), "%.12g", static_cast<double>(v));
466 }
467 return buf;
468 }

484 // Parses a whole number text; false if empty or not a number.
485 static bool parseNumber(const std::string& text, float& out) {
486 if (text.empty() || text == "-" || text == ".") return false;
487 char* end = nullptr;
488 out = strtof(text.c_str(), &end);
489 return end != nullptr && *end == '\0';
490 }

492 // Keys of the plot pad. The From/To fields report focus through onFocus.
493 void handlePlotKey(const std::string& a) {
494 if (a == "back") {
495 closePlot();
496 return;
497 }
498 if (a == "from") {
499 fromField_.focus();
500 return;
501 }
502 if (a == "to") {
503 toField_.focus();
504 return;
505 }
506 std::string& text = activeField_ == 0 ? fromText_ : toText_;
507 if (a == "del") {
508 if (!text.empty()) text.erase(text.size() - 1);
509 } else if (a == "-") {
510 if (text.empty()) text = "-"; // a minus sign only at the start
511 } else if (a == ".") {
512 if (text.find('.') == std::string::npos) text += ".";
513 } else if (a.size() == 1 && a[0] >= '0' && a[0] <= '9') {
514 if (text.size() < 9) text += a;
515 } else {
516 return;
517 }
518 fromField_.setText(fromText_.c_str());
519 toField_.setText(toText_.c_str());
520 applyRangeText();
521 }
522
523 // Redraws the plot if both fields hold a valid range (To above From).
524 void applyRangeText() {
525 float lo = 0.0f, hi = 0.0f;
526 if (!parseNumber(fromText_, lo) || !parseNumber(toText_, hi) || hi <= lo) return;
527 plotMin_ = lo;
528 plotMax_ = hi;
529 samplePlot(); // the expression compiled when the plot opened
530 }
531
532 // Samples the expression as a function of x over [plotMin_, plotMax_] and
533 // shows it in the chart. Points where the expression is undefined become
534 // NaN, which the chart draws as a gap.
535 Error samplePlot() {
536 typename Calc::Program prog; // compiled once, run for every sample
537 Error e = calc_.compile(expr_.c_str(), prog);
538 if (e != Error::None) return e;
539 samples_.assign(kPlotSamples, nanValue());
540 for (size_t i = 0; i < kPlotSamples; ++i) {
541 double x = plotMin_ + (plotMax_ - plotMin_) * static_cast<double>(i) /
542 static_cast<double>(kPlotSamples - 1);
543 calc_.setVariable("x", static_cast<Number>(x));
544 typename Calc::Result r = calc_.run(prog);
545 if (r.ok()) samples_[i] = static_cast<float>(r.value);
546 }
547 calc_.setVariable("x", Number(0)); // leave x at 0 for the keypad
548
549 char footer[48];
550 snprintf(footer, sizeof(footer), "x from %g to %g", static_cast<double>(plotMin_),
551 static_cast<double>(plotMax_));
552 std::string title = "f(x) = " + expr_;
553 plot_.setTitle(title.c_str());
554 plot_.setFooter(footer);
555 plot_.setXAxis(plotMin_, plotMax_, 5);
556 plot_.setValues(0, samples_);
557 return Error::None;
558 }
