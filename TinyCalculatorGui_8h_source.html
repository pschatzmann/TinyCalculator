207 void press(const char* action) {
208 if (action == nullptr || *action == '\0') return;
209 std::string a = action;
210 error_.clear();
211
212 if (plotting_) {
213 handlePlotKey(a);
214 refresh();
215 return;
216 }
217
218 if (a == "page") {
219 changePage(1); // to the next enabled page
220 } else if (a == "plot") {
221 plotExpression();
222 } else if (a == "angle") {
223 toggleAngle();
224 } else if (a == "clear") {
225 expr_.clear();
226 showResult_ = false;
227 } else if (a == "del") {
228 if (showResult_) {
229 expr_.clear();
230 showResult_ = false;
231 } else if (!expr_.empty()) {
232 expr_.erase(expr_.size() - 1);
233 }
234 } else if (a == "=") {
235 evaluate();
236 } else {
237 // digits, operators, functions and constants
238 // after "=", an operator continues with the result, anything else starts over
239 std::string prefix = expr_;
240 if (showResult_) {
241 bool continues = a == "+" || a == "-" || a == "*" || a == "/" || a == "^" ||
242 a == "%" || a == "!" || a == "," || a == "to";
243 prefix = continues ? "ans" : "";
244 }
245 if (!acceptable(a, prefix)) { // invalid combination: the key is ignored
246 refresh();
247 return;
248 }
249 if (showResult_) {
250 expr_ = prefix;
251 showResult_ = false;
252 }
253 // "to" is a conversion, ":=" starts a body: both are written out in full
254 const std::string token = a == "to" ? " to " : a == "def" ? "=" : a;
255 if (needsProduct(expr_, token)) expr_ += "*";
256 expr_ += token;
257 }
258 refresh();
259 }

261 // ---- key entry rules --------------------------------------------------
262 // The keypad only accepts keys that continue the expression validly, e.g.
263 // no operator after an operator, no ")" without "(", no second "." in a
264 // number. Rejected keys are ignored.

266 static char lastChar(const std::string& t) { return t.empty() ? '\0' : t[t.size() - 1]; }
267 static bool isDigitChar(char c) { return c >= '0' && c <= '9'; }
268 static bool isLetter(char c) { return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z'); }
269 static bool isOperatorChar(char c) {
270 return c == '+' || c == '-' || c == '*' || c == '/' || c == '^' || c == '%' || c == ',';
271 }
272 static bool isBinaryOp(const std::string& a) {
273 return a == "+" || a == "*" || a == "/" || a == "^" || a == "%";
274 }
275 static bool isNameKey(const std::string& a) {
276 return a == "pi" || a == "e" || a == "tau" || a == "phi" || a == "ans" || a == "x" ||
277 isUnitName(a);
278 }

280 // The units (see unitTable() in Functions.h).
281 static bool isUnitName(const std::string& a) {
282 return a == "m" || a == "cm" || a == "mm" || a == "km" || a == "mi" || a == "ft" ||
283 a == "in" || a == "yd" || a == "kg" || a == "g" || a == "lb" || a == "oz" ||
284 a == "s" || a == "min" || a == "h" || a == "day" || a == "degC" ||
285 a == "degF" || a == "K";
286 }
287 static bool isFunctionKey(const std::string& a) { return !a.empty() && a[a.size() - 1] == '('; }
288
289 // number of '(' not yet closed
290 static int openDepth(const std::string& t) {
291 int depth = 0;
292 for (char c : t) {
293 if (c == '(') ++depth;
294 if (c == ')' && depth > 0) --depth;
295 }
296 return depth;
297 }

299 // true if the number being typed already contains a '.'
300 static bool numberHasDot(const std::string& t) {
301 for (size_t i = t.size(); i-- > 0;) {
302 if (t[i] == '.') return true;
303 if (!isDigitChar(t[i])) return false;
304 }
305 return false;
306 }

308 // true if key `a` may follow the expression `t`
309 // Dimension of a unit: 0 none, 1 length, 2 mass, 3 time, 4 temperature.
310 static int unitDimension(const std::string& u) {
311 if (u == "m" || u == "cm" || u == "mm" || u == "km" || u == "mi" || u == "ft" ||
312 u == "in" || u == "yd") {
313 return 1;
314 }
315 if (u == "kg" || u == "g" || u == "lb" || u == "oz") return 2;
316 if (u == "s" || u == "min" || u == "h" || u == "day") return 3;
317 if (u == "degC" || u == "degF" || u == "K") return 4;
318 return 0;
319 }
320
321 // The letters at the end of t (the name just typed), e.g. "km" in "5 km".
322 static std::string trailingWord(const std::string& t) {
323 size_t b = t.size();
324 while (b > 0 && isLetter(t[b - 1])) --b;
325 return t.substr(b);
326 }

328 // Conversion state of an expression: a "to" is in it, the target unit is
329 // still missing (awaiting), and the source unit before the "to".
330 struct ToState {
331 bool present;
332 bool awaiting;
333 std::string source;
334 };
335
336 static ToState toState(const std::string& t) {
337 ToState st{false, false, ""};
338 const size_t k = t.rfind(" to");
339 if (k == std::string::npos) return st;
340 const size_t after = k + 3;
341 if (after < t.size() && t[after] != ' ') return st; // "tox" is not the keyword
342 st.present = true;
343 st.awaiting = t.find_first_not_of(' ', after) == std::string::npos;
344 st.source = trailingWord(t.substr(0, k));
345 return st;
346 }
347
348 // True if the unit key may be pressed for this expression.
349 static bool unitAllowed(const std::string& unit, const std::string& t) {
350 const ToState st = toState(t);
351 if (!st.present) return true; // a unit can always be typed before "to"
352 return st.awaiting && unitDimension(unit) != 0 &&
353 unitDimension(unit) == unitDimension(st.source);
354 }
355
356 static bool acceptable(const std::string& a, const std::string& t) {
357 const ToState conv = toState(t);
358 if (conv.present) { // after "to": only a unit of the same kind as the source
359 return conv.awaiting && isUnitName(a) && unitAllowed(a, t);
360 }
361 if (a == "to") return unitDimension(trailingWord(t)) != 0; // a unit comes first
362 const char c = lastChar(t);
363 const bool empty = t.empty();
364 const bool open = c == '(';
365 const bool op = isOperatorChar(c);
366 // a number, name, ")" or "!" (a space after "to" is not an operand yet)
367 const bool operand = !empty && !open && !op && c != ' ';
368
369 if (a == "(" || isFunctionKey(a)) return true; // a function is also a call start
370 if (a == ")") return openDepth(t) > 0 && operand;
371 if (a == "!") return operand && c != '.';
372 if (a == ",") return openDepth(t) > 0 && operand;
373 if (a == "-") return c != '-'; // subtraction or a sign, but never "--"
374 if (a == ".") return !numberHasDot(t) && !isLetter(c) && c != ')' && c != '!';
375 if (isBinaryOp(a) || a == "to") return operand;
376 if (a == "def") return openDepth(t) == 0 && c == ')'; // a function header "f(x)" then "="
377 if (isNameKey(a)) return c != '.';
378 return true; // digits
379 }
380
381 // an explicit "*" is needed when a name, ")" or "!" is followed by an operand
382 // (so that "pi" then "2" is "pi*2", not the name "pi2")
383 static bool needsProduct(const std::string& t, const std::string& a) {
384 const char c = lastChar(t);
385 const bool endsOperand = isLetter(c) || c == ')' || c == '!';
386 const bool startsOperand = isDigitChar(a[0]) || isLetter(a[0]) || a[0] == '(';
387 return endsOperand && startsOperand;
388 }

587 static const int32_t kMargin = 8; // outer margin
588 static const int32_t kGap = 4; // gap between keys and fields
589 static const int32_t kExprH = 20; // expression line (at most)
590 static const int32_t kFieldH = 34; // From / To fields (at most)

592 // Positions all widgets for the current size (bounds_) and mode.
593 //
594 // Wide areas (landscape) put the display on the left and the keys on the
595 // right; tall areas (portrait) stack the display above the keys. The keys
596 // use the space the display does not need, so the keypad fills the area.
597 void relayout() {
598 panel_.bounds = bounds_;
599 const int32_t x = bounds_.x, y = bounds_.y, w = bounds_.w, h = bounds_.h;
600 const bool landscape = w > h;
601 tinymd::Bounds display, pad;
602 if (landscape) {
603 const int32_t dw = (w - 2 * kMargin) / 2;
604 display = tinymd::Bounds(x + kMargin, y + kMargin, dw, h - 2 * kMargin);
605 pad = tinymd::Bounds(x + kMargin + dw + kGap, y + kMargin,
606 w - 2 * kMargin - dw - kGap, h - 2 * kMargin);
607 } else {
608 // the plot needs more height than the expression and result lines
609 const int32_t dh = plotting_ ? (h * 11) / 20 : (h * 3) / 10;
610 display = tinymd::Bounds(x + kMargin, y + kMargin, w - 2 * kMargin, dh);
611 pad = tinymd::Bounds(x + kMargin, y + kMargin + dh + kGap,
612 w - 2 * kMargin, h - 2 * kMargin - dh - kGap);
613 }
614
615 if (plotting_) {
616 const int32_t fh = min32(kFieldH, display.h / 4);
617 // room for the field labels ("From", "To"), which are drawn above the fields
618 const int32_t labelH = 18;
619 plot_.bounds = tinymd::Bounds(display.x, display.y, display.w,
620 display.h - fh - labelH - kGap);
621 const int32_t fw = (display.w - kGap) / 2;
622 const int32_t fy = display.y + display.h - fh;
623 fromField_.bounds = tinymd::Bounds(display.x, fy, fw, fh);
624 toField_.bounds = tinymd::Bounds(display.x + fw + kGap, fy, display.w - fw - kGap, fh);
625 placeKeys(pad, 4); // numeric pad
626 } else {
627 const int32_t eh = min32(kExprH, display.h / 3);
628 exprLabel_.bounds = tinymd::Bounds(display.x, display.y, display.w, eh);
629 resultLabel_.bounds = tinymd::Bounds(display.x, display.y + eh, display.w,
630 display.h - eh);
631 placeKeys(pad, 7); // keypad
632 }
633 }

637 // Places the key buttons in a grid of `rows` rows (4 columns) inside `area`.
638 void placeKeys(tinymd::Bounds area, int32_t rows) {
639 const int32_t cols = 4;
640 int32_t keyW = (area.w - (cols - 1) * kGap) / cols;
641 int32_t keyH = (area.h - (rows - 1) * kGap) / rows;
642 if (keyW < 1) keyW = 1;
643 if (keyH < 1) keyH = 1;
644 for (size_t i = 0; i < kKeyCount; ++i) {
645 keys_[i].onClick = [this, i]() { press(activeKeys()[i].action); };
646 int32_t row = static_cast<int32_t>(i) / cols;
647 int32_t col = static_cast<int32_t>(i) % cols;
648 if (row >= rows) {
649 // not on the grid: zero size, so hidden keys do not extend the scroll area
650 keys_[i].bounds = tinymd::Bounds(bounds_.x, bounds_.y, 0, 0);
651 continue;
652 }
653 keys_[i].bounds = tinymd::Bounds(area.x + col * (keyW + kGap),
654 area.y + row * (keyH + kGap), keyW, keyH);
655 }
656 }

687 static tinymd::ButtonVariant variantFor(const char* action) {
688 std::string a = action;

704 void evaluate() {
705 if (expr_.empty()) return;
706 if (Calc::isDefinition(expr_.c_str())) { // "f(x) = ...": defines a function
707 typename Calc::Result r = calc_.evaluate(expr_.c_str());
708 showResult_ = r.ok();
709 if (r.ok()) {
710 resultText_ = expr_.substr(0, expr_.find('(')) + " defined";
711 } else {
712 error_ = errorText(r.error);
713 }
714 return;
715 }
716 typename Calc::Result r = calc_.evaluate(expr_.c_str()); // also updates "ans"
717 if (r.ok()) {
718 resultText_ = format(r.value);
719 showResult_ = true;
720 } else {
721 showResult_ = false;
722 error_ = errorText(r.error);
723 }
724 }

751 // Parses a whole number text; false if empty or not a number.
752 static bool parseNumber(const std::string& text, float& out) {
753 if (text.empty() || text == "-" || text == ".") return false;
754 char* end = nullptr;
755 out = strtof(text.c_str(), &end);
756 return end != nullptr && *end == '\0';

759 // Keys of the plot pad. The From/To fields report focus through onFocus.
760 void handlePlotKey(const std::string& a) {
761 if (a == "back") {
762 closePlot();
763 return;
764 }
765 if (a == "from") {
766 fromField_.focus();
767 return;
768 }
769 if (a == "to") {
770 toField_.focus();
771 return;
772 }
773 std::string& text = activeField_ == 0 ? fromText_ : toText_;
774 if (a == "del") {
775 if (!text.empty()) text.erase(text.size() - 1);
776 } else if (a == "-") {
777 if (text.empty()) text = "-"; // a minus sign only at the start
778 } else if (a == ".") {
779 if (text.find('.') == std::string::npos) text += ".";
780 } else if (a.size() == 1 && a[0] >= '0' && a[0] <= '9') {
781 if (text.size() < 9) text += a;
782 } else {
783 return;
784 }
785 fromField_.setText(fromText_.c_str());
786 toField_.setText(toText_.c_str());
787 applyRangeText();
788 }

790 // Redraws the plot if both fields hold a valid range (To above From).
791 void applyRangeText() {
792 float lo = 0.0f, hi = 0.0f;
793 if (!parseNumber(fromText_, lo) || !parseNumber(toText_, hi) || hi <= lo) return;
794 plotMin_ = lo;
795 plotMax_ = hi;
796 samplePlot(); // the expression compiled when the plot opened
797 }

799 // Samples the expression as a function of x over [plotMin_, plotMax_] and
800 // shows it in the chart. Points where the expression is undefined become
801 // NaN, which the chart draws as a gap.
802 Error samplePlot() {
803 typename Calc::Program prog; // compiled once, run for every sample
804 Error e = calc_.compile(expr_.c_str(), prog);
805 if (e != Error::None) return e;
806 samples_.assign(kPlotSamples, nanValue());
807 for (size_t i = 0; i < kPlotSamples; ++i) {
808 double x = plotMin_ + (plotMax_ - plotMin_) * static_cast<double>(i) /
809 static_cast<double>(kPlotSamples - 1);
810 calc_.setVariable("x", static_cast<Number>(x));
811 typename Calc::Result r = calc_.run(prog);
812 if (r.ok()) samples_[i] = static_cast<float>(r.value);
813 }
814 calc_.setVariable("x", Number(0)); // leave x at 0 for the keypad
815
816 char footer[48];
817 snprintf(footer, sizeof(footer), "x from %g to %g", static_cast<double>(plotMin_),
818 static_cast<double>(plotMax_));
819 std::string title = "f(x) = " + expr_;
820 plot_.setTitle(title.c_str());
821 plot_.setFooter(footer);
822 plot_.setXAxis(plotMin_, plotMax_, 5);
823 plot_.setValues(0, samples_);
824 return Error::None;
825 }

857 // The enabled page `step` steps away (step = +1 or -1); the Basic page is always enabled.
858 size_t nextPage(int step) const {
859 const int pages = static_cast<int>(kPageCount);
860 size_t candidate = page_;
861 for (int i = 0; i < pages; ++i) {
862 candidate = static_cast<size_t>((static_cast<int>(candidate) + (step < 0 ? -1 : 1) + pages) % pages);
863 if (candidate == 0 || (pages_ & (1u << candidate)) != 0) return candidate;
864 }
865 return 0;
866 }

951 // Mixes colour c towards `toward` by a (0 = c, 1 = toward).
952 static RGB_T fade(RGB_T c, RGB_T toward, float a) {
953 auto mix = [a](int x, int y) {
954 return static_cast<uint8_t>(x + (y - x) * a + 0.5f);
955 };
956 return RGB_T::fromRGB(mix(c.getRed(), toward.getRed()), mix(c.getGreen(), toward.getGreen()),
957 mix(c.getBlue(), toward.getBlue()));
958 }

963 CalculatorTheme<RGB_T> d = t;
