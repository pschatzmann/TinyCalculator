90 void press(const char* action) {
91 if (action == nullptr || *action == '\0') return;
92 std::string a = action;
93 error_.clear();
94
95 if (a == "page") {
96 page_ = (page_ + 1) % kPageCount;
97 refreshKeys();
98 } else if (a == "angle") {
99 toggleAngle();
100 } else if (a == "clear") {
101 expr_.clear();
102 showResult_ = false;
103 } else if (a == "del") {
104 if (showResult_) {
105 expr_.clear();
106 showResult_ = false;
107 } else if (!expr_.empty()) {
108 expr_.erase(expr_.size() - 1);
109 }
110 } else if (a == "=") {
111 evaluate();
112 } else {
113 // digits, operators, functions and constants
114 if (showResult_) {
115 // an operator after "=" continues with the result
116 bool continues = a == "+" || a == "-" || a == "*" || a == "/" || a == "^" ||
117 a == "%" || a == "!" || a == ",";
118 expr_ = continues ? "ans" : "";
119 showResult_ = false;
120 }
121 expr_ += a;
122 }
123 refresh();
124 }

197 // Position the widgets inside `b`: expression line, result line, keypad.
198 void layout(tinymd::Bounds b) {
199 const int32_t m = 8; // outer margin
200 const int32_t g = 4; // gap between keys
201 const int32_t exprH = 20, resultH = 44;
202 const int32_t top = b.y + m + exprH + resultH + g; // keypad top
203 const int32_t cols = 4, rows = 7;
204 const int32_t keyW = (b.w - 2 * m - (cols - 1) * g) / cols;
205 const int32_t keyH = (b.y + b.h - m - top - (rows - 1) * g) / rows;
206
207 exprLabel_.bounds = tinymd::Bounds(b.x + m, b.y + m, b.w - 2 * m, exprH);
208 resultLabel_.bounds = tinymd::Bounds(b.x + m, b.y + m + exprH, b.w - 2 * m, resultH);
209
210 for (size_t i = 0; i < kKeyCount; ++i) {
211 int32_t row = static_cast<int32_t>(i) / cols;
212 int32_t col = static_cast<int32_t>(i) % cols;
213 keys_[i].bounds = tinymd::Bounds(b.x + m + col * (keyW + g),
214 top + row * (keyH + g), keyW, keyH);
215 keys_[i].onClick = [this, i]() { press(keys(page_)[i].action); };
216 }
217 }

219 // Shows the key table of the current page.
220 void refreshKeys() {
221 const KeyDef* defs = keys(page_);
222 for (size_t i = 0; i < kKeyCount; ++i) {
223 const bool used = defs[i].action != nullptr && *defs[i].action != '\0';
224 keys_[i].visible = used;
225 if (!used) continue;
226 keys_[i].setLabel(labelOf(defs[i]));
227 keys_[i].setVariant(variantFor(defs[i].action));
228 }
229 }

231 const char* labelOf(const KeyDef& d) const {
232 if (strcmp(d.action, "page") == 0) return pageName(page_);
233 if (strcmp(d.action, "angle") == 0) {
234 return calc_.angleUnit() == AngleUnit::Degrees ? "DEG" : "RAD";

239 static tinymd::ButtonVariant variantFor(const char* action) {
240 std::string a = action;
241 if (a == "=") return tinymd::ButtonVariant::kFilled;
242 if (a == "angle" || a == "page" || a == "clear" || a == "del") {
243 return tinymd::ButtonVariant::kText;
244 }
245 if (a.size() == 1 && (a[0] >= '0' && a[0] <= '9')) return tinymd::ButtonVariant::kOutlined;
246 if (a == ".") return tinymd::ButtonVariant::kOutlined;
247 return tinymd::ButtonVariant::kTonal;
248 }

256 void evaluate() {
257 if (expr_.empty()) return;
258 typename Calc::Result r = calc_.evaluate(expr_.c_str()); // also updates "ans"
259 if (r.ok()) {
260 resultText_ = format(r.value);
261 showResult_ = true;
262 } else {
263 showResult_ = false;
264 error_ = errorText(r.error);
265 }
266 }

268 static std::string format(Number v) {
269 // 7 significant digits for float, 12 for double
270 char buf[32];
271 if (sizeof(Number) <= 4) {
272 snprintf(buf, sizeof(buf), "%.7g", static_cast<double>(v));
273 } else {
274 snprintf(buf, sizeof(buf), "%.12g", static_cast<double>(v));
275 }
276 return buf;
277 }

279 void refresh() {
280 std::string e = showResult_ ? expr_ + " =" : expr_;

282 resultLabel_.setText(resultText().c_str());
283 if (screen_ != nullptr) screen_->invalidate();
284 }
