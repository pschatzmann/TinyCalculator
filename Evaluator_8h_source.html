31// Gamma for x that is not a pole; integers use an exact product.
32template <typename T>
33Error gammaChecked(T x, T& out) {
34 typedef Math<T> M;
35 if (M::isInteger(x)) {
36 if (x <= T()) return Error::Domain;
37 if (x > M::c(171)) return Error::Overflow;
38 T r = M::c(1);
39 for (int i = 2; static_cast<T>(i) < x; ++i) r *= static_cast<T>(i);
40 out = r;
41 return Error::None;
42 }
43 out = M::gamma(x);
44 return Error::None;
45}

47// Evaluates one built-in function. `a` points to n >= minArgs arguments.
48template <typename T>
49Error callFunction(Fn fn, const T* a, size_t n, AngleUnit unit, T& out) {
50 typedef Math<T> M;
51 typedef Prim<T> P;
52 const bool deg = unit == AngleUnit::Degrees;
53 const T x = a[0];
54 const T y = n > 1 ? a[1] : T();
55 const T zero = M::c(0);
56 const T one = M::c(1);
57 T r = zero;
58
59 switch (fn) {
60 case Fn::Abs: r = M::absolute(x); break;
61 case Fn::Sqrt:
62 if (x < zero) return Error::Domain;
63 r = P::sqrt(x);
64 break;
65 case Fn::Cbrt: r = M::cbrt(x); break;
66 case Fn::Root: // x^(1/y), real odd roots of negative numbers allowed
67 if (y == zero) return Error::DivisionByZero;
68 if (x < zero) {
69 if (!M::isInteger(y) || P::fmod(y, M::c(2)) == zero) return Error::Domain;
70 r = -P::pow(-x, one / y);
71 } else if (x == zero) {
72 if (y < zero) return Error::DivisionByZero;
73 r = zero;
74 } else {
75 r = P::pow(x, one / y);
76 }
77 break;
78 case Fn::Exp: r = P::exp(x); break;
79 case Fn::Exp2: r = M::exp2(x); break;
80 case Fn::Expm1: r = M::expm1(x); break;
81 case Fn::Ln:
82 if (x <= zero) return Error::Domain;
83 r = P::log(x);
84 break;
85 case Fn::Log: // log(x) = log10(x), log(x, base)
86 if (x <= zero) return Error::Domain;
87 if (n == 1) {
88 r = P::log10(x);
89 } else {
90 if (y <= zero || y == one) return Error::Domain;
91 r = P::log(x) / P::log(y);
92 }
93 break;
94 case Fn::Log2:
95 if (x <= zero) return Error::Domain;
96 r = M::log2(x);
97 break;
98 case Fn::Log10:
99 if (x <= zero) return Error::Domain;
100 r = P::log10(x);
101 break;
102 case Fn::Pow: return powChecked(x, y, out);
103
104 case Fn::Sin: r = P::sin(toRadians(x, deg)); break;
105 case Fn::Cos: r = P::cos(toRadians(x, deg)); break;
106 case Fn::Tan: r = P::tan(toRadians(x, deg)); break;
107 case Fn::Asin:
108 if (M::absolute(x) > one) return Error::Domain;
109 r = fromRadians(P::asin(x), deg);
110 break;
111 case Fn::Acos:
112 if (M::absolute(x) > one) return Error::Domain;
113 r = fromRadians(P::acos(x), deg);
114 break;
115 case Fn::Atan: r = fromRadians(P::atan(x), deg); break;
116 case Fn::Atan2: // atan2(y, x) with the first argument as y
117 if (x == zero && y == zero) return Error::Domain;
118 r = fromRadians(P::atan2(x, y), deg);
119 break;
120
121 case Fn::Sinh: r = M::sinh(x); break;
122 case Fn::Cosh: r = M::cosh(x); break;
123 case Fn::Tanh: r = M::tanh(x); break;
124 case Fn::Asinh: r = M::asinh(x); break;
125 case Fn::Acosh:
126 if (x < one) return Error::Domain;
127 r = M::acosh(x);
128 break;
129 case Fn::Atanh:
130 if (M::absolute(x) >= one) return Error::Domain;
131 r = M::atanh(x);
132 break;
133
134 case Fn::Sec: {
135 T c = P::cos(toRadians(x, deg));
136 if (c == zero) return Error::DivisionByZero;
137 r = one / c;
138 break;
139 }
140 case Fn::Csc: {
141 T s = P::sin(toRadians(x, deg));
142 if (s == zero) return Error::DivisionByZero;
143 r = one / s;
144 break;
145 }
146 case Fn::Cot: {
147 T s = P::sin(toRadians(x, deg));
148 if (s == zero) return Error::DivisionByZero;
149 r = P::cos(toRadians(x, deg)) / s;
150 break;
151 }
152 case Fn::Asec:
153 if (M::absolute(x) < one) return Error::Domain;
154 r = fromRadians(P::acos(one / x), deg);
155 break;
156 case Fn::Acsc:
157 if (M::absolute(x) < one) return Error::Domain;
158 r = fromRadians(P::asin(one / x), deg);
159 break;
160 case Fn::Acot:
161 r = x == zero ? fromRadians(M::pi() / M::c(2), deg)
162 : fromRadians(P::atan(one / x), deg);
163 break;
164
165 case Fn::Floor: r = P::floor(x); break;
166 case Fn::Ceil: r = P::ceil(x); break;
167 case Fn::Round: r = M::roundHalf(x); break;
168 case Fn::Trunc: r = M::trunc(x); break;
169 case Fn::Frac: r = M::frac(x); break;
170 case Fn::Sign: r = M::sign(x); break;
171 case Fn::Mod: // floored modulo: result has the sign of y
172 if (y == zero) return Error::DivisionByZero;
173 r = x - y * P::floor(x / y);
174 break;
175 case Fn::Hypot: r = M::hypot(x, y); break;
176
177 case Fn::Gamma: return gammaChecked(x, out);
178 case Fn::Fact: return gammaChecked(x + one, out); // x! = gamma(x + 1)
179 case Fn::Erf: r = M::erf(x); break;
180 case Fn::Erfc: r = M::erfc(x); break;
181
182 case Fn::Comb:
183 case Fn::Perm: { // n choose k, n permute k
184 if (!M::isNonNegativeInteger(x) || !M::isNonNegativeInteger(y) || y > x) {
185 return Error::Domain;
186 }
187 if (x > M::c(1000000.0)) return Error::Overflow;
188 int nn = static_cast<int>(x);
189 int k = static_cast<int>(y);
190 r = one;
191 if (fn == Fn::Comb) {
192 for (int i = 1; i <= k; ++i) {
193 r = r * static_cast<T>(nn - k + i) / static_cast<T>(i);
194 }
195 } else {
196 for (int i = 0; i < k; ++i) r *= static_cast<T>(nn - i);
197 }
198 break;
199 }
200 case Fn::Gcd:
201 case Fn::Lcm: {
202 if (!M::isInteger(x) || !M::isInteger(y)) return Error::Domain;
203 if (M::absolute(x) > M::c(2147483647.0) || M::absolute(y) > M::c(2147483647.0)) {
204 return Error::Domain;
205 }
206 uint32_t a1 = static_cast<uint32_t>(M::absolute(x));
207 uint32_t b1 = static_cast<uint32_t>(M::absolute(y));
208 uint32_t p = a1, q = b1;
209 while (q != 0) { // Euclid
210 uint32_t t = p % q;
211 p = q;
212 q = t;
213 }
214 if (fn == Fn::Gcd) {
215 r = static_cast<T>(p);
216 } else {
217 r = p == 0 ? zero : (M::absolute(x) / static_cast<T>(p)) * M::absolute(y);
218 }
219 break;
220 }
221
222 case Fn::Min:
223 case Fn::Max:
224 case Fn::Sum:
225 case Fn::Avg: {
226 T acc = a[0];
227 for (size_t i = 1; i < n; ++i) {
228 if (fn == Fn::Min) {
229 if (a[i] < acc) acc = a[i];
230 } else if (fn == Fn::Max) {
231 if (a[i] > acc) acc = a[i];
232 } else {
233 acc += a[i];
234 }
235 }
236 r = fn == Fn::Avg ? acc / M::c(static_cast<double>(n)) : acc;
237 break;
238 }
239 case Fn::Clamp: {
240 T hi = a[2];
241 if (y > hi) return Error::Domain;
242 r = x < y ? y : (x > hi ? hi : x);
243 break;
244 }
245 case Fn::Deg: r = x * (M::c(180) / M::pi()); break;
246 case Fn::Rad: r = x * (M::pi() / M::c(180)); break;
247 default:
248 return Error::UnknownFunction;
249 }
250 out = r;
251 return Error::None;
252}

254template <typename T>
255Error applyBinary(OpCode code, T a, T b, T& out) {
256 switch (code) {
257 case OpCode::Add: out = a + b; return Error::None;
258 case OpCode::Sub: out = a - b; return Error::None;
259 case OpCode::Mul: out = a * b; return Error::None;
260 case OpCode::Div:
261 if (b == T()) return Error::DivisionByZero;
262 out = a / b;
263 return Error::None;
264 case OpCode::Mod: // C-style remainder (sign of dividend)
265 if (b == T()) return Error::DivisionByZero;
266 out = Prim<T>::fmod(a, b);
267 return Error::None;
268 case OpCode::Pow: return powChecked(a, b, out);
269 default: return Error::Syntax;
270 }
271}

285// Runs a compiled program. `frame` holds the parameters of the running user
286// function (none at the top level).
287template <typename T, size_t N>
288Error evaluateProgram(const Expression<T, N>& ex, const Environment<T, N>& env,
289 const T* frame, size_t frameCount, T& result) {
290 typedef Math<T> M;
291 T stack[N];
292 size_t sp = 0;
293 for (size_t i = 0; i < ex.size(); ++i) {
294 const Op<T>& op = ex[i];
295 Error err = Error::None;
296 switch (op.code) {
297 case OpCode::Push:
298 stack[sp++] = op.value;
299 break;
300 case OpCode::Load:
301 if (op.index & kFrameBit) { // a parameter of a user function
302 const size_t k = op.index & ~kFrameBit;
303 if (k >= frameCount) return Error::UnknownIdentifier;
304 stack[sp++] = frame[k];
305 } else {
306 if (op.index >= env.varCount) return Error::UnknownIdentifier;
307 stack[sp++] = env.vars[op.index];
308 }
309 break;
310 case OpCode::Neg:
311 stack[sp - 1] = -stack[sp - 1];
312 break;
313 case OpCode::Call: {
314 sp -= op.argc;
315 T r = T();
316 err = callFunction(static_cast<Fn>(op.index), stack + sp, op.argc, env.unit, r);
317 stack[sp++] = r;
318 break;
319 }
320 case OpCode::UserCall: {
321 if (op.index >= env.functionCount) return Error::UnknownFunction;
322 if (env.depth >= TINYCALC_MAX_FUNCTIONS) return Error::NestingTooDeep;
323 sp -= op.argc;
324 Environment<T, N> inner = env;
325 inner.depth = env.depth + 1;
326 T r = T();
327 err = evaluateProgram(env.functions[op.index], inner, stack + sp, op.argc, r);
328 stack[sp++] = r;
329 break;
330 }
331 default: { // binary operators
332 --sp;
333 T r = T();
334 err = applyBinary(op.code, stack[sp - 1], stack[sp], r);
335 stack[sp - 1] = r;
336 break;
337 }
338 }
339 if (err != Error::None) return err;
340 if (!M::isFinite(stack[sp - 1])) {
341 return M::isNaN(stack[sp - 1]) ? Error::Domain : Error::Overflow;
342 }
343 }
344 if (sp != 1) return Error::Syntax;
345 result = stack[0];
346 return Error::None;
347}
