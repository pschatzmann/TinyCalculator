166 // Detects "identifier = rest" (but not "==").
167 static bool splitAssignment(const char* text, char* target, const char*& rhs) {
168 const char* p = text;
169 while (detail::isSpace(*p)) ++p;
170 if (!detail::isIdentStart(p[0])) return false;
171 size_t len = 0;
172 while (detail::isIdentPart(p[len])) ++len;
173 if (len > TINYCALC_MAX_NAME) return false;
174 memcpy(target, p, len);
175 target[len] = '\0';
176 const char* q = p + len;
177 while (detail::isSpace(*q)) ++q;
178 if (*q != '=' || q[1] == '=') return false;
179 rhs = q + 1;
180 return true;
181 }
