127 // ---- user-defined functions -------------------------------------------
128 // Defines (or redefines) a function from text of the form "name(a, b) = body".
129 // The body may use the parameters, variables, constants, built-in functions
130 // and functions defined before it. A function cannot call itself.
131 Error define(const char* text) {
132 const char* p = text;
133 skipSpaces(p);
134 char name[TINYCALC_MAX_NAME + 1];
135 if (!parseName(p, name)) return Error::InvalidName;
136 skipSpaces(p);
137 if (*p != '(') return Error::Syntax;
138 ++p;
139
140 char params[kMaxParams][TINYCALC_MAX_NAME + 1];
141 size_t np = 0;
142 skipSpaces(p);
143 if (*p != ')') {
144 for (;;) {
145 skipSpaces(p);
146 if (np >= kMaxParams || !parseName(p, params[np])) return Error::InvalidName;
147 T dummy = T();
148 if (lookupConstant<T>(params[np], dummy) || isFunctionName(params[np])) {
149 return Error::ReservedName;
150 }
151 for (size_t k = 0; k < np; ++k) {
152 if (strcmp(params[k], params[np]) == 0) return Error::InvalidName; // duplicate
153 }
154 ++np;
155 skipSpaces(p);
156 if (*p != ',') break;
157 ++p;
158 }
159 }
160 if (*p != ')') return Error::Syntax;
161 ++p;
162 skipSpaces(p);
163 if (*p != '=' || p[1] == '=') return Error::Syntax;
164 const char* body = p + 1;
165
166 if (isReserved(name)) return Error::ReservedName;
167 int slot = userFunctionIndex(name); // an existing function is redefined
168 if (slot < 0) slot = static_cast<int>(fnCount_); // else a new one, if there is room
169 BodyResolver resolver = {this, params, np, slot};
170 Program compiled;
171 size_t pos = 0;
172 detail::Compiler<T, MaxOps, BodyResolver> compiler(body, compiled, resolver);
173 Error e = compiler.run(pos);
174 if (e != Error::None) return e;
175 if (static_cast<size_t>(slot) >= kMaxFunctions) return Error::TooManyFunctions;
176
177 fnBodies_[slot] = compiled;
178 strncpy(fnNames_[slot], name, TINYCALC_MAX_NAME);
179 fnNames_[slot][TINYCALC_MAX_NAME] = '\0';
180 fnParams_[slot] = static_cast<uint8_t>(np);
181 if (static_cast<size_t>(slot) == fnCount_) ++fnCount_;
182 return Error::None;
183 }

206 // Evaluates "expr", "name = expr" or a function definition "f(x) = expr".
207 // Sets "ans" on success. A definition has no value.
208 Result evaluate(const char* text) {
209 if (isDefinition(text)) return makeResult(T(), define(text), 0);
210 bool converted = false;
211 Result conv = temperatureConversion(text, converted);
212 if (converted) {
213 if (conv.ok()) values_[0] = conv.value; // ans
214 return conv;
215 }
216 char target[TINYCALC_MAX_NAME + 1];
217 const char* expr = text;
218 bool assign = splitAssignment(text, target, expr);
219
220 Result r = evaluateExpression(expr);
221 if (!r.ok()) {
222 r.errorPos += static_cast<size_t>(expr - text);
223 return r;
224 }
225 if (assign) {
226 Error e = setVariable(target, r.value);
227 if (e != Error::None) return makeResult(T(), e, 0);
228 }
229 values_[0] = r.value; // ans
230 return r;
231 }

268 static bool parseName(const char*& p, char* out) {

286 // Temperature scale: kelvin = value * scale + offset.
287 static bool temperatureScale(const char* unit, T& scale, T& offset) {
288 switch (temperatureIndex(unit)) {
289 case 0: // degC
290 scale = T(1);
291 offset = T(273.15);
292 return true;
293 case 1: // degF
294 scale = T(0.5555555555555556); // 5/9
295 offset = T(255.3722222222222); // 459.67 * 5/9
296 return true;
297 case 2: // K
298 scale = T(1);
299 offset = T(0);
300 return true;
301 default:
302 return false;
303 }

315 // Recognises "<value> <from> to <to>" with temperature units. Sets handled
316 // to true if the text has this form and converts it.
317 Result temperatureConversion(const char* text, bool& handled) {
318 handled = false;
319 const char* p = text + strlen(text);
320 while (p > text && detail::isSpace(p[-1])) --p;
321 const char* toEnd = p;
322 while (p > text && detail::isIdentStart(p[-1])) --p; // letters: "100degC" is 100 degC
323 const char* toStart = p;
324 char to[TINYCALC_MAX_NAME + 1];
325 if (!copyRange(toStart, toEnd, to, sizeof(to)) || !isTemperatureUnit(to)) {
326 return makeResult(T(), Error::None, 0);
327 }
328 // the keyword "to"
329 while (p > text && detail::isSpace(p[-1])) --p;
330 if (p - text < 2 || p[-2] != 't' || p[-1] != 'o') return makeResult(T(), Error::None, 0);
331 const char* keyword = p - 2;
332 if (keyword > text && detail::isIdentPart(keyword[-1])) return makeResult(T(), Error::None, 0);
333 p = keyword;
334 while (p > text && detail::isSpace(p[-1])) --p;
335 const char* fromEnd = p;
336 while (p > text && detail::isIdentStart(p[-1])) --p;
337 char from[TINYCALC_MAX_NAME + 1];
338 if (!copyRange(p, fromEnd, from, sizeof(from))) return makeResult(T(), Error::None, 0);
339 handled = true;
340 if (!isTemperatureUnit(from)) return makeResult(T(), Error::Syntax, 0); // no source unit
341 const char* valueEnd = p;
342 while (valueEnd > text && detail::isSpace(valueEnd[-1])) --valueEnd;
343 if (valueEnd == text) return makeResult(T(), Error::Syntax, 0); // no value
344 char value[kMaxValueText];
345 if (!copyRange(text, valueEnd, value, sizeof(value))) {
346 return makeResult(T(), Error::TooComplex, 0);
347 }
348 T s1 = T(), o1 = T(), s2 = T(), o2 = T();
349 temperatureScale(from, s1, o1);
350 temperatureScale(to, s2, o2);
351 Result v = evaluateExpression(value);
352 if (!v.ok()) return v;
353 const T kelvin = v.value * s1 + o1;
354 return makeResult((kelvin - o2) / s2, Error::None, 0);
355 }

399 // Detects "identifier = rest" (but not "==").
400 static bool splitAssignment(const char* text, char* target, const char*& rhs) {
401 const char* p = text;
402 while (detail::isSpace(*p)) ++p;
403 if (!detail::isIdentStart(p[0])) return false;
404 size_t len = 0;
405 while (detail::isIdentPart(p[len])) ++len;
406 if (len > TINYCALC_MAX_NAME) return false;
407 memcpy(target, p, len);
408 target[len] = '\0';
409 const char* q = p + len;
410 while (detail::isSpace(*q)) ++q;
411 if (*q != '=' || q[1] == '=') return false;
412 rhs = q + 1;
413 return true;
414 }
